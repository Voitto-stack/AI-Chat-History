 SELECT                          
  COALESCE(SUM(amount::numeric),  
  0)::float AS total_amount       
  FROM user_call_order            
  WHERE (male_user_id = 7038987 OR
   female_user_id = 7038987)      
    AND create_at >=              
  '2026-04-18'::date              
    AND create_at <
  '2026-04-19'::date
