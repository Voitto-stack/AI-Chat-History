SELECT
    COALESCE(call_total, 0) +
  COALESCE(ins_total, 0) AS       
  grand_total,
    call_total,                   
    ins_total,    
    video_call_amount,
    voice_call_amount,
    mock_video_amount,
    video_call_count,
    voice_call_count,
    mock_video_count,
    ins_count
  FROM (                          
    -- 通话订单（按类型分组）
    SELECT                        
      SUM(CAST(NULLIF(amount, '')
  AS NUMERIC)) AS call_total,     
      SUM(CASE WHEN order_type =
  'VIDEO_CALL' THEN               
  CAST(NULLIF(amount, '') AS
  NUMERIC) ELSE 0 END) AS         
  video_call_amount,
      SUM(CASE WHEN order_type =
  'VOICE_CALL' THEN               
  CAST(NULLIF(amount, '') AS
  NUMERIC) ELSE 0 END) AS         
  voice_call_amount,
      SUM(CASE WHEN order_type =
  'MOCK_VIDEO' THEN               
  CAST(NULLIF(amount, '') AS
  NUMERIC) ELSE 0 END) AS         
  mock_video_amount,
      COUNT(*) FILTER (WHERE
  order_type = 'VIDEO_CALL') AS
  video_call_count,
      COUNT(*) FILTER (WHERE
  order_type = 'VOICE_CALL') AS   
  voice_call_count,
      COUNT(*) FILTER (WHERE      
  order_type = 'MOCK_VIDEO') AS
  mock_video_count
    FROM user_call_order
    WHERE female_user_id = 7038987
      AND status = 'PAID'
      AND create_at >=            
  '2026-04-18'::date              
      AND create_at <
  '2026-04-19'::date              
  ) call_stats,   
  (                               
    -- INS 交换订单               
    SELECT                        
      SUM(CAST(NULLIF(uieb.balance
  _change, '') AS NUMERIC)) AS    
  ins_total,
      COUNT(*) AS ins_count       
    FROM user_ins_exchange_order
  uieo                            
    LEFT JOIN
  pwa_user_balance_change_history 
  uieb            
      ON uieo.id =                
  uieb.ins_exchange_order_id
    WHERE uieo.female_user_id =
  7038987                         
      AND uieo.status = 'FINISH'
      AND uieo.create_at >=       
  '2026-04-18'::date
      AND uieo.create_at <
  '2026-04-19'::date              
  ) ins_stats;
