SELECT
    SUM(CAST(NULLIF(amount, '') AS
   NUMERIC)) AS video_call_income,
    COUNT(*) AS video_call_count,
    SUM(call_duration) AS         
  video_call_duration             
  FROM user_call_order            
  WHERE female_user_id = 7038987  
    AND status = 'PAID'           
    AND order_type = 'VIDEO_CALL'
    AND create_at >=              
  '2026-04-18'::date              
    AND create_at <               
  '2026-04-19'::date;
