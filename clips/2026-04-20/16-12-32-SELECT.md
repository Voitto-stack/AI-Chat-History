    u.user_id,    
    u.username,
    u.nickname,

    CASE u.app_name

      WHEN 3 THEN 'HAVEN_PWA'

      WHEN 5 THEN 'WAVE'

      WHEN 10 THEN 'POKA'

      WHEN 13 THEN 'LUMA'

      WHEN 16 THEN 'DONI_IOS'

    u.age,
    u.gender,
    u.cai_user_type,
    u.regulation_status,
    u.platform,

    u.face_score,

    u.created_at,                                                                                                                                                                       
    u.updated_at,                                                                                                                                                                       
    u.deleted_at                                                                                                                                                                        
  FROM userinfo u 
  WHERE u.user_id = 2100046452                                                                                                                                                          
    AND u.deleted_at IS NULL;
